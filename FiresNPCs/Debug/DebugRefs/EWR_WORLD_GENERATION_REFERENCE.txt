// Helper.cs — Math utilities for altitude/height conversion
// =============================================================================

public static class Helper
{
    // Valheim's height system:
    // - "altitude" = height relative to water level
    // - "height" = altitude + water level (absolute Y)
    // - "base height" = height / 200f (normalized for WorldGenerator)

    public static float AltitudeToBaseHeight(float altitude)
    {
        return HeightToBaseHeight(AltitudeToHeight(altitude));
    }

    public static float HeightToBaseHeight(float altitude)
    {
        return altitude / 200f;
    }

    public static float AltitudeToHeight(float altitude)
    {
        return Configuration.WaterLevel + altitude;
    }

    // Transpiler helpers for replacing IL constants:

    // Replace(instructions, oldSbyte, newSbyte)

    // ReplaceSeed(instructions, fieldName, newSeedValue)
}

// =============================================================================
// Generate.cs — Triggers world regeneration
// =============================================================================

public static void World()
{
    Patcher.Patch();
    if (WorldGenerator.instance == null) return;

    EWR.Log.LogInfo("Regenerating the world.");
    WorldGenerator.instance.Pregenerate();

    // Regenerate ALL heightmaps (terrain chunks)
    foreach (Heightmap heightmap in Object.FindObjectsOfType<Heightmap>())
    {
        heightmap.m_buildData = null;
        heightmap.Regenerate();
    }

    // Clear clutter (grass etc.)
    ClutterSystem.instance?.ClearAll();

    // Restart all water volumes
    foreach (WaterVolume waterVolume in WaterHelper.Get())
        waterVolume.Start();

    // Regenerate minimap if enabled
    if (Configuration.RegenerateMap)
        Map();
}

public static void Map()
{
    if (SystemInfo.graphicsDeviceType == GraphicsDeviceType.Null) return;
    Minimap.instance?.GenerateWorldMap();
}
