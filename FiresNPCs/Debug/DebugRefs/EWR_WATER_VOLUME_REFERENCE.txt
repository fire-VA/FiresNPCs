// =============================================================================
// WaterHelper.cs — Central water level & wave management
// =============================================================================

public class WaterHelper
{
    public static Vector3 OriginalScale = Vector3.zero;

    // Find all active WaterVolume objects in scene
    public static WaterVolume[] Get() => Object.FindObjectsOfType<WaterVolume>();

    // Set water level on a WaterVolume (only global wind volumes = ocean/lakes)
    public static void SetLevel(WaterVolume obj)
    {
        if (!obj.m_useGlobalWind) return;

        // Move the WaterVolume transform to configured water level
        Vector3 position1 = obj.transform.position;
        position1.y = Configuration.WaterLevel;
        obj.transform.position = position1;

        // Also move the water surface child object
        Vector3 position2 = obj.m_waterSurface.transform.position;
        position2.y = Configuration.WaterLevel;
        obj.m_waterSurface.transform.position = position2;
    }

    // Set water level on ZoneSystem (affects zone loading water check)
    public static void SetLevel(ZoneSystem obj)
    {
        if (obj == null) return;
        obj.m_waterLevel = Configuration.WaterLevel;
    }

    // Set water level on ClutterSystem (affects grass near water)
    public static void SetLevel(ClutterSystem obj)
    {
        if (obj == null) return;
        obj.m_waterLevel = Configuration.WaterLevel;
    }

    // Modify wave size on WaterVolume
    public static void SetWaveSize(WaterVolume obj)
    {
        if (!obj.m_useGlobalWind) return;

        // Capture original scale on first call
        if (OriginalScale == Vector3.zero)
            OriginalScale = obj.m_waterSurface.transform.localScale;

        Vector3 scale = OriginalScale;
        if (Configuration.WaveOnlyHeight)
            scale.y *= Configuration.WaveMultiplier;  // Only scale Y (height)
        else
            scale *= Configuration.WaveMultiplier;    // Scale all axes

        obj.m_waterSurface.transform.localScale = scale;
    }
}

// RIVER RENDERING IN VANILLA:
// Rivers/streams don't use WaterVolume directly.
// Instead, WorldGenerator stores river data and:
// 1. Heightmap.Generate() checks WorldGenerator.GetRiverWeight() for each vertex
// 2. If weight > 0, terrain is LOWERED to create the river bed
// 3. The lowered terrain sits below m_waterLevel, so ocean water fills the gap
// 4. River "width" controls how wide the terrain depression is
// 5. The "curveWidth" and "curveWavelength" create the meandering path
//
// KEY METHODS IN VANILLA WorldGenerator:
//   GetRiverWeight(float x, float z) -> returns 0-1 river influence
//   Uses m_riverPoints spatial cache for fast lookup
//   RiverPoint has position + width — terrain is lowered proportional to width

// SO RIVERS IN VANILLA ARE:
// - NOT separate water objects
// - Just terrain depressions that sit below the global water level
// - The ocean water mesh naturally fills them
// - River "rendering" = the ocean mesh + terrain depression
// - River "physics" = terrain shape + ocean WaterVolume

// THIS IS IMPORTANT FOR OUR IMPLEMENTATION:
// Our custom water system creates ACTUAL water meshes (WaterMeshCombiner).
// We can either:
// A) Modify terrain heightmap + use vanilla water (like EWR does)
// B) Create terrain depressions + place our WaterDisk objects along the path
// C) Only place WaterDisk objects (no terrain mod — water floats above terrain)
//
// Option B is most natural for our system:
// - Use the river path data to calculate positions
// - Place WaterDisks along the path at appropriate heights
// - Let our WaterMeshCombiner handle the mesh generation
// - Optionally modify terrain to create the river bed
